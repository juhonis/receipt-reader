cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(receipt_ocr_worker LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
find_package(Leptonica CONFIG REQUIRED)
include("${TESSERACT_BUILD}/TesseractTargets.cmake")
add_executable(receipt_ocr_worker worker.cpp)
target_compile_features(receipt_ocr_worker PRIVATE cxx_std_17)
target_include_directories(receipt_ocr_worker PRIVATE "${TESSERACT_BUILD}/include" ${Leptonica_INCLUDE_DIRS})
target_compile_options(receipt_ocr_worker PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(receipt_ocr_worker PRIVATE Tesseract::libtesseract)
